constant addSlopeRegionFunc

添加坡度坡向分析区域

Example
var pts = [lon1,lat1,h1,lon2,lat2,h2,lon3,lat3,h3];
 var slope2 = new SuperMapCesiumPlugin.SlopeSetting();
        slope2.DisplayMode = SuperMapCesiumPlugin.SlopeSettingEnum.DisplayMode.FACE_AND_ARROW;
        slope2.MaxVisibleValue = 78;
        slope2.MinVisibleValue = 0;
        var colorTable = new SuperMapCesiumPlugin.ColorTable();
        colorTable.insert(80, new Cesium.Color(255 / 255, 0 / 255, 0 / 255));
        colorTable.insert(50, new Cesium.Color(221 / 255, 224 / 255, 7 / 255));
        colorTable.insert(30, new Cesium.Color(20 / 255, 187 / 255, 18 / 255));
        colorTable.insert(20, new Cesium.Color(0, 161 / 255, 1));
        colorTable.insert(0, new Cesium.Color(9 / 255, 9 / 255, 255 / 255));
        var wide = SuperMapCesiumPlugin.HypsometricSettingEnum.AnalysisRegionMode.ARM_REGION;
        slope2.ColorTable = colorTable;
        slope2.CoverageArea = pts;
        viewer.scene.globe.addSlopeRegion({
            name : 'test',
            slopeSetting: slope2,
            analysisMode: wide
        });
